{{/* 漢字／臺羅平行文本（對齊邏輯同 taigi_typing） 用法一：內文用單獨一行 --- 分隔（可多行）；可選第三段華語對譯（辭典式括號顯示在下方） {{< tailo_parallel >}} 漢字稿 --- 臺羅稿 --- 華語稿（選填） {{< /tailo_parallel >}} 用法二：具名參數（mandarin / caption 選填） {{< tailo_parallel hanji="…" tailo="…" mandarin="…" caption="…" >}} default：預設顯示模式 hanji | tailo | split（split＝漢字稿、臺羅稿兩段原始文本） */}} {{- $hanji := "" -}} {{- $tailo := "" -}} {{- $mandarin := "" -}} {{- $caption := "" -}} {{- $default := default "hanji" (.Get "default") -}} {{- $allowedModes := slice "hanji" "tailo" "split" -}} {{- if not (in $allowedModes $default) }}{{ $default = "hanji" }}{{ end -}} {{- $inner := replaceRE `\r\n` "\n" .Inner | strings.TrimSpace -}} {{- $hk := .Get "hanji" -}} {{- $tk := .Get "tailo" -}} {{- $mdParam := .Get "mandarin" -}} {{- with .Get "caption" }}{{ $caption = . | strings.TrimSpace }}{{ end -}} {{- if and $hk $tk -}} {{- $hanji = $hk | strings.TrimSpace -}} {{- $tailo = $tk | strings.TrimSpace -}} {{- with $mdParam }}{{ $mandarin = . | strings.TrimSpace }}{{ end -}} {{- else if $inner -}} {{- $parts := split $inner "\n---\n" -}} {{- $n := len $parts -}} {{- if or (lt $n 2) (gt $n 3) -}} {{- errorf "shortcode tailo_parallel: 內文須為 2 段（漢字稿---臺羅稿）或 3 段（再多一個---華語稿）；或使用 hanji= / tailo=（目前 %d 段）" $n -}} {{- end -}} {{- $hanji = index $parts 0 | strings.TrimSpace -}} {{- $tailo = index $parts 1 | strings.TrimSpace -}} {{- if eq $n 3 }}{{ $mandarin = index $parts 2 | strings.TrimSpace }}{{ end -}} {{- else -}} {{- errorf "shortcode tailo_parallel: 請提供 hanji 與 tailo 參數，或使用內文 + --- 分隔" -}} {{- end -}} {{- if not (.Page.Scratch.Get "taigiParallelJsIncluded") -}} {{- .Page.Scratch.Set "taigiParallelJsIncluded" true -}} {{- end -}}
